# Build index (k = 25)

# Quantify each sample (non-strand-specific mode; do NOT add --rf-stranded)
kallisto quant -i results/justAS_k25.idx -o results/quant/LD1 -t 8 R1_LD1.fastq.gz R2_LD1.fastq.gz
kallisto quant -i results/justAS_k25.idx -o results/quant/LD2 -t 8 R1_LD2.fastq.gz R2_LD2.fastq.gz
kallisto quant -i results/justAS_k25.idx -o results/quant/D1  -t 8 R1_D1.fastq.gz  R2_D1.fastq.gz
kallisto quant -i results/justAS_k25.idx -o results/quant/D2  -t 8 R1_D2.fastq.gz  R2_D2.fastq.gz
